## 7.1 结果分析概述

### 7.1.1 结果分析的目标

结果分析帮助我们从评估数据中提取有价值的信息：

```
BFCLItemEvalResult (原始结果)
    ↓
统计分析 (延迟、Token、成本)
    ↓
准确率评估 (函数匹配、参数匹配)
    ↓
对比分析 (不同模型对比)
    ↓
可视化 (图表、报告)
```

💡 **分析维度**：
- **性能指标**：延迟、Token 使用量、成本
- **准确率**：函数调用准确率、参数提取准确率
- **对比分析**：不同模型、不同配置对比
- **趋势分析**：性能趋势、错误模式

import json
from pathlib import Path
import statistics

def load_results(result_dir):
    """加载所有结果文件"""
    results = []
    result_path = Path(result_dir)
    
    for file in result_path.glob("*_result.json"):
        with open(file, 'r') as f:
            for line in f:
                results.append(json.loads(line))
    
    return results

def analyze_results(results):
    """基础统计分析"""
    print("=== 基础统计 ===\n")
    
    # 总数
    print(f"总测试项数: {len(results)}")
    
    # 延迟统计
    latencies = [r['latency'] for r in results]
    print(f"\n延迟统计:")
    print(f"  最小值: {min(latencies):.2f}s")
    print(f"  最大值: {max(latencies):.2f}s")
    print(f"  平均值: {statistics.mean(latencies):.2f}s")
    print(f"  中位数: {statistics.median(latencies):.2f}s")
    print(f"  P95:    {sorted(latencies)[int(len(latencies)*0.95)]:.2f}s")
    
    # Token 统计
    input_tokens = [r['input_token_count'] for r in results]
    output_tokens = [r['output_token_count'] for r in results]
    
    print(f"\nToken 统计:")
    print(f"  输入 Token 总数: {sum(input_tokens)}")
    print(f"  输出 Token 总数: {sum(output_tokens)}")
    print(f"  平均输入 Token:  {statistics.mean(input_tokens):.0f}")
    print(f"  平均输出 Token:  {statistics.mean(output_tokens):.0f}")
    
    # 成本估算（以 GPT-4 为例）
    input_cost = sum(input_tokens) * 0.03 / 1000  # $0.03/1K tokens
    output_cost = sum(output_tokens) * 0.06 / 1000  # $0.06/1K tokens
    total_cost = input_cost + output_cost
    
    print(f"\n成本估算 (GPT-4):")
    print(f"  输入成本: ${input_cost:.2f}")
    print(f"  输出成本: ${output_cost:.2f}")
    print(f"  总成本:   ${total_cost:.2f}")

import pandas as pd

def group_analysis(results):
    """按类别分组统计"""
    df = pd.DataFrame(results)
    
    # 从 ID 提取类别
    df['category'] = df['id'].apply(lambda x: x.split('_')[0])
    
    # 按类别分组统计
    grouped = df.groupby('category').agg({
        'latency': ['mean', 'median', 'min', 'max'],
        'input_token_count': 'sum',
        'output_token_count': 'sum',
        'id': 'count'
    }).round(2)
    
    print("=== 按类别分组统计 ===\n")
    print(grouped)
    
    # 计算每个类别的成本
    grouped['cost'] = (
        grouped[('input_token_count', 'sum')] * 0.03 / 1000 +
        grouped[('output_token_count', 'sum')] * 0.06 / 1000
    ).round(2)
    
    print("\n=== 每个类别的成本 ===\n")
    print(grouped['cost'])

## 7.4 准确率评估

### 7.4.1 函数调用准确率

```python
def evaluate_function_call_accuracy(results, ground_truth):
    """
    评估函数调用准确率
    
    Args:
        results: 评估结果列表
        ground_truth: 真实标签字典 {id: expected_function_calls}
    """
    correct = 0
    total = len(results)
    
    detailed_results = []
    
    for result in results:
        item_id = result['id']
        predicted = result['result']
        expected = ground_truth.get(item_id, [])
        
        # 比较函数调用
        is_correct = compare_function_calls(predicted, expected)
        
        if is_correct:
            correct += 1
        
        detailed_results.append({
            'id': item_id,
            'correct': is_correct,
            'predicted': predicted,
            'expected': expected
        })
    
    accuracy = correct / total if total > 0 else 0
    
    print("=== 函数调用准确率 ===")
    print(f"正确: {correct}/{total}")
    print(f"准确率: {accuracy:.2%}")
    
    return accuracy, detailed_results

def compare_function_calls(predicted, expected):
    """比较函数调用是否匹配"""
    if len(predicted) != len(expected):
        return False
    
    # 排序后比较
    predicted_sorted = sorted(predicted, key=lambda x: list(x.keys())[0])
    expected_sorted = sorted(expected, key=lambda x: list(x.keys())[0])
    
    for pred, exp in zip(predicted_sorted, expected_sorted):
        pred_func = list(pred.keys())[0]
        exp_func = list(exp.keys())[0]
        
        if pred_func != exp_func:
            return False
        
        # 比较参数
        pred_args = pred[pred_func]
        exp_args = exp[exp_func]
        
        if pred_args != exp_args:
            return False
    
    return True
```

💡 **评估逻辑**：
1. 比较函数调用数量
2. 比较函数名称
3. 比较函数参数

### 7.4.2 参数提取准确率

```python
def evaluate_parameter_accuracy(results, ground_truth):
    """评估参数提取准确率"""
    total_params = 0
    correct_params = 0
    
    for result in results:
        item_id = result['id']
        predicted = result['result']
        expected = ground_truth.get(item_id, [])
        
        for pred, exp in zip(predicted, expected):
            pred_func = list(pred.keys())[0]
            exp_func = list(exp.keys())[0]
            
            if pred_func != exp_func:
                continue
            
            pred_args = pred[pred_func]
            exp_args = exp[exp_func]
            
            # 比较每个参数
            for key in exp_args.keys():
                total_params += 1
                if key in pred_args and pred_args[key] == exp_args[key]:
                    correct_params += 1
    
    accuracy = correct_params / total_params if total_params > 0 else 0
    
    print("=== 参数提取准确率 ===")
    print(f"正确参数: {correct_params}/{total_params}")
    print(f"准确率: {accuracy:.2%}")
    
    return accuracy
```

💡 **评估逻辑**：
1. 遍历所有函数调用
2. 比较每个参数的值
3. 计算参数级别的准确率

```python
def analyze_errors(detailed_results):
    """详细错误分析"""
    errors = [r for r in detailed_results if not r['correct']]
    
    print(f"\n=== 错误分析 ===")
    print(f"总错误数: {len(errors)}")
    
    # 错误类型统计
    error_types = {
        'function_mismatch': 0,
        'parameter_mismatch': 0,
        'missing_function': 0,
        'extra_function': 0
    }
    
    for error in errors:
        predicted = error['predicted']
        expected = error['expected']
        
        if len(predicted) == 0:
            error_types['missing_function'] += 1
        elif len(predicted) > len(expected):
            error_types['extra_function'] += 1
        else:
            # 检查函数名是否匹配
            pred_funcs = set(list(p.keys())[0] for p in predicted)
            exp_funcs = set(list(e.keys())[0] for e in expected)
            
            if pred_funcs != exp_funcs:
                error_types['function_mismatch'] += 1
            else:
                error_types['parameter_mismatch'] += 1
    
    print("\n错误类型分布:")
    for error_type, count in error_types.items():
        print(f"  {error_type}: {count}")
    
    # 显示前 5 个错误示例
    print("\n前 5 个错误示例:")
    for i, error in enumerate(errors[:5], 1):
        print(f"\n{i}. ID: {error['id']}")
        print(f"   预测: {error['predicted']}")
        print(f"   期望: {error['expected']}")
```

💡 **分析要点**：
- **错误分类**：识别不同类型的错误
- **错误示例**：查看具体错误案例
- **改进方向**：根据错误类型制定改进策略

def compare_models(result_dirs):
    """
    对比多个模型的评估结果
    
    Args:
        result_dirs: 字典 {model_name: result_dir}
    """
    all_results = {}
    
    for model_name, result_dir in result_dirs.items():
        results = load_results(result_dir)
        all_results[model_name] = results
    
    # 创建对比表格
    comparison_data = []
    
    for model_name, results in all_results.items():
        latencies = [r['latency'] for r in results]
        input_tokens = [r['input_token_count'] for r in results]
        output_tokens = [r['output_token_count'] for r in results]
        
        comparison_data.append({
            '模型': model_name,
            '测试项数': len(results),
            '平均延迟': f"{statistics.mean(latencies):.2f}s",
            'P95 延迟': f"{sorted(latencies)[int(len(latencies)*0.95)]:.2f}s",
            '输入 Token': sum(input_tokens),
            '输出 Token': sum(output_tokens),
            '总 Token': sum(input_tokens) + sum(output_tokens),
        })
    
    df = pd.DataFrame(comparison_data)
    print("=== 模型对比 ===\n")
    print(df.to_string(index=False))
    
    # 可视化对比
    plt.figure(figsize=(14, 5))
    
    # 1. 延迟对比
    plt.subplot(1, 3, 1)
    for model_name, results in all_results.items():

    print(df.to_string(index=False))
    
    # 可视化
    plt.figure(figsize=(8, 5))

    plt.xlabel('模型')
    plt.ylabel('准确率')
    plt.title('模型准确率对比')
    plt.ylim(0, 1)
    plt.grid(True, alpha=0.3, axis='y')
    
    for i, v in enumerate(accuracies):
        plt.text(i, v + 0.02, f"{v:.1%}", ha='center')
    
    plt.tight_layout()
    plt.savefig('accuracy_comparison.png', dpi=150)
    plt.show()
    
    return df

```python
def generate_markdown_report(results, output_file='report.md'):
    """生成 Markdown 格式报告"""
    
    latencies = [r['latency'] for r in results]
    input_tokens = [r['input_token_count'] for r in results]
    output_tokens = [r['output_token_count'] for r in results]
    
    report = f"""# Benchmark Report

## 概览

- **测试项数**: {len(results)}
- **测试日期**: {pd.Timestamp.now().strftime('%Y-%m-%d %H:%M:%S')}

## 性能指标

### 延迟

| 指标 | 值 |
|------|-----|
| 最小值 | {min(latencies):.2f}s |
| 最大值 | {max(latencies):.2f}s |
| 平均值 | {statistics.mean(latencies):.2f}s |
| 中位数 | {statistics.median(latencies):.2f}s |
| P95 | {sorted(latencies)[int(len(latencies)*0.95)]:.2f}s |

### Token 使用

| 指标 | 值 |
|------|-----|
| 输入 Token 总数 | {sum(input_tokens)} |
| 输出 Token 总数 | {sum(output_tokens)} |
| 平均输入 Token | {statistics.mean(input_tokens):.0f} |
| 平均输出 Token | {statistics.mean(output_tokens):.0f} |

### 成本估算

| 指标 | 值 |
|------|-----|
| 输入成本 | ${sum(input_tokens) * 0.03 / 1000:.2f} |
| 输出成本 | ${sum(output_tokens) * 0.06 / 1000:.2f} |
| 总成本 | ${(sum(input_tokens) * 0.03 + sum(output_tokens) * 0.06) / 1000:.2f} |

## 详细结果

### 延迟 Top 10

| 排名 | ID | 延迟 |
|------|-----|------|
"""
    
    # 添加延迟 Top 10
    sorted_results = sorted(results, key=lambda x: x['latency'], reverse=True)
    for i, result in enumerate(sorted_results[:10], 1):
        report += f"| {i} | {result['id']} | {result['latency']:.2f}s |\n"
    
    report += """
## 结论

本报告展示了 benchmark 测试的详细结果，包括性能指标和成本分析。

### 关键发现

1. **延迟表现**: 平均延迟为 {:.2f}s，P95 延迟为 {:.2f}s
2. **Token 效率**: 平均每个测试项消耗 {} 个 Token
3. **成本控制**: 总成本为 ${:.2f}

### 建议

- 优化高延迟的测试项
- 考虑使用更经济的模型
- 监控 Token 使用趋势
""".format(
        statistics.mean(latencies),
        sorted(latencies)[int(len(latencies)*0.95)],
        int(statistics.mean(input_tokens) + statistics.mean(output_tokens)),
        (sum(input_tokens) * 0.03 + sum(output_tokens) * 0.06) / 1000
    )
    
    # 保存报告
    with open(output_file, 'w') as f:
        f.write(report)
    
    print(f"Report saved to: {output_file}")

# 使用示例
generate_markdown_report(results, 'benchmark_report.md')
```

💡 **报告内容**：
- 概览信息
- 性能指标（延迟、Token、成本）
- Top 10 高延迟项
- 结论和建议
