```go
    Field   string
    Message string
}

func (e *ValidationError) Error() string {
    return fmt.Sprintf("validation error: %s - %s", e.Field, e.Message)
}

func validateRequest(req ExecuteRequest) error {

func process() {
    data, _ := os.ReadFile("file.txt")  // 忽略错误

}

// ❌ 避免：panic

    if err := doSomething(); err != nil {
        panic(err)  // 不要在生产代码中使用 panic
```

```go
// ✅ 推荐：严格的输入验证
func validatePath(path string) error {
    // 1. 检查长度
    if len(path) > 1024 {
        return fmt.Errorf("path too long")
    }
    
    // 2. 清理路径
    cleanPath := filepath.Clean(path)
    
    // 3. 检查目录遍历
    if strings.Contains(cleanPath, "..") {
        return fmt.Errorf("directory traversal not allowed")
    }
    
    // 4. 检查允许的目录
    allowedDirs := []string{"/workspace", "/tmp"}
    allowed := false
    for _, dir := range allowedDirs {
        if strings.HasPrefix(cleanPath, dir) {
            allowed = true
            break
        }
    }
    
    if !allowed {
        return fmt.Errorf("path not in allowed directories")
    }
    
    return nil
}

// ❌ 避免：不验证输入
func readFile(path string) ([]byte, error) {
    return os.ReadFile(path)  // 没有验证 path
}
```

```go
// ✅ 推荐：使用 goroutine 池
type WorkerPool struct {
    tasks chan func()
    wg    sync.WaitGroup
}

func NewWorkerPool(size int) *WorkerPool {
    pool := &WorkerPool{
        tasks: make(chan func(), 100),
    }
    
    for i := 0; i < size; i++ {
        pool.wg.Add(1)
        go func() {
            defer pool.wg.Done()
            for task := range pool.tasks {
                task()
            }
        }()
    }
    
    return pool
}

func (p *WorkerPool) Submit(task func()) {
    p.tasks <- task
}

// 使用

defer pool.Close()

pool.Submit(func() {
    // 执行任务
})
```

### 13.5.2 缓存策略

```go
// ✅ 推荐：使用缓存减少重复计算
type Cache struct {
    mu    sync.RWMutex
    items map[string]*CacheItem
}

type CacheItem struct {
    Value     interface{}
    ExpiresAt time.Time
}

func (c *Cache) Get(key string) (interface{}, bool) {
    c.mu.RLock()
    defer c.mu.RUnlock()
    
    item, exists := c.items[key]
    if !exists || time.Now().After(item.ExpiresAt) {
        return nil, false
    }
    
    return item.Value, true
}

func (c *Cache) Set(key string, value interface{}, ttl time.Duration) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    c.items[key] = &CacheItem{
        Value:     value,
        ExpiresAt: time.Now().Add(ttl),
    }
}

// 使用
cache := &Cache{items: make(map[string]*CacheItem)}

// 缓存 MCP 服务器列表
servers, found := cache.Get("mcp_servers")
if !found {
    servers = loadMcpServers()
    cache.Set("mcp_servers", servers, 5*time.Minute)
}
```

```go
// ✅ 推荐：流式处理大文件
func handleFileDownload(w http.ResponseWriter, path string) error {
    file, err := os.Open(path)
    if err != nil {
        return err
    }
    defer file.Close()
    
    // 设置响应头
    stat, _ := file.Stat()
    w.Header().Set("Content-Length", strconv.FormatInt(stat.Size(), 10))
    w.Header().Set("Content-Type", "application/octet-stream")
    
    // 流式传输，不占用大量内存
    buf := make([]byte, 32*1024)  // 32KB 缓冲区
    for {
        n, err := file.Read(buf)
        if n > 0 {
            if _, writeErr := w.Write(buf[:n]); writeErr != nil {
                return writeErr
            }
        }
        if err == io.EOF {
            break
        }
        if err != nil {
            return err
        }
    }
    
    return nil
}
```

```go
// ✅ 推荐：添加性能指标
type Metrics struct {
    RequestCount    *prometheus.CounterVec
    RequestDuration *prometheus.HistogramVec
    ActiveTasks     prometheus.Gauge
}

var metrics *Metrics

func initMetrics() {
    metrics = &Metrics{
        RequestCount: prometheus.NewCounterVec(
            prometheus.CounterOpts{
                Name: "sandbox_requests_total",
                Help: "Total number of requests",
            },
            []string{"method", "path", "status"},
        ),
        RequestDuration: prometheus.NewHistogramVec(
            prometheus.HistogramOpts{
                Name:    "sandbox_request_duration_seconds",
                Help:    "Request duration in seconds",
                Buckets: prometheus.DefBuckets,
            },
            []string{"method", "path"},
        ),
        ActiveTasks: prometheus.NewGauge(
            prometheus.GaugeOpts{
                Name: "sandbox_active_tasks",
                Help: "Number of active tasks",
            },
        ),
    }
    
    prometheus.MustRegister(metrics.RequestCount)
    prometheus.MustRegister(metrics.RequestDuration)
    prometheus.MustRegister(metrics.ActiveTasks)
}

// 在中间件中使用
func metricsMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        
        wrapped := &responseWriter{ResponseWriter: w, statusCode: 200}
        next.ServeHTTP(wrapped, r)
        
        duration := time.Since(start).Seconds()
        metrics.RequestCount.WithLabelValues(r.Method, r.URL.Path, strconv.Itoa(wrapped.statusCode)).Inc()
        metrics.RequestDuration.WithLabelValues(r.Method, r.URL.Path).Observe(duration)
    })
}
```

## 13.10 常见问题与解决方案

### 13.10.1 内存泄漏

**问题**：内存使用持续增长

**原因**：
- 未关闭的文件句柄
- 未清理的 goroutine
- 缓存未设置过期时间

**解决方案**：
```go
// ✅ 正确：使用 defer 关闭资源
func processFile(path string) error {
    file, err := os.Open(path)
    if err != nil {
        return err
    }
    defer file.Close()  // 确保关闭
    
    // 处理文件
    return nil
}

// ✅ 正确：使用 context 控制 goroutine 生命周期
func processWithTimeout(ctx context.Context) error {
    ctx, cancel := context.WithTimeout(ctx, 5*time.Second)
    defer cancel()
    
    select {
    case result := <-doWork():
        return result
    case <-ctx.Done():
        return ctx.Err()
    }
}

// ✅ 正确：缓存设置过期时间
cache.Set(key, value, 5*time.Minute)
```

### 13.10.2 性能瓶颈

**问题**：响应时间过长

**原因**：
- 同步阻塞操作
- 未使用缓存
- 数据库查询慢

**解决方案**：
```go
// ✅ 使用异步处理
func handleRequest(w http.ResponseWriter, r *http.Request) {
    // 异步执行耗时操作
    go func() {
        result := expensiveOperation()
        // 通过 channel 或回调通知完成
    }()
    
    // 立即返回响应
    w.WriteHeader(http.StatusAccepted)
}

// ✅ 使用缓存
func getData(key string) (string, error) {
    // 先查缓存
    if value, found := cache.Get(key); found {
        return value.(string), nil
    }
    
    // 缓存未命中，查询数据库
    value, err := db.Query(key)
    if err != nil {
        return "", err
    }
    
    // 写入缓存
    cache.Set(key, value, 5*time.Minute)
    
    return value, nil
}
```

### 13.10.3 安全漏洞

**问题**：发现路径遍历漏洞

**原因**：
- 未验证用户输入
- 路径清理不完整

**解决方案**：
```go
// ✅ 严格的路径验证
func validatePath(path string) error {
    // 1. 清理路径
    cleanPath := filepath.Clean(path)
    
    // 2. 检查目录遍历
    if strings.Contains(cleanPath, "..") {
        return fmt.Errorf("directory traversal not allowed")
    }
    
    // 3. 检查允许的目录
    allowedDirs := []string{"/workspace", "/tmp"}
    for _, dir := range allowedDirs {
        if strings.HasPrefix(cleanPath, dir) {
            return nil
        }
    }
    
    return fmt.Errorf("path not in allowed directories")
}

// ✅ 使用绝对路径
func resolvePath(base, relative string) (string, error) {
    absPath := filepath.Join(base, relative)
    cleanPath := filepath.Clean(absPath)
    
    // 确保在基础目录内
    if !strings.HasPrefix(cleanPath, base) {
        return "", fmt.Errorf("path escapes base directory")
    }
    
    return cleanPath, nil
}
```
